# Lab Store reservation stack (from W4 Ch10 Round 1, reference answer).
#
# Access and usage are different questions. Allow/Disallow say where a crawler
# may go. The usage lines say what anyone who fetched a page may do with it.
# Search stays open: the search crawler keeps indexing, and ai-train is the
# signal it honours. Training is *licensed*, not forbidden: a verified crawler
# that pays the RSL price (see /license.xml) is served; one that does not is
# answered with 402, never 403.

User-agent: *
Allow: /
Disallow: /account
Disallow: /admin/
Disallow: /lab/
Disallow: /maze/
Content-Signal: search=yes, ai-input=no, ai-train=no
Content-Usage: train-ai=n, ai-use=n, search=y

# A scraping intermediary that declares itself gets no access at all. It will
# ignore this, and the log line that proves it ignored this is the point.
User-agent: ScrapeFeed-Bot
Disallow: /
