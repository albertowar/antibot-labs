# Reference reservation stack (instructor solution, not the only good one).
#
# Crawl access and usage preferences are different questions. Access rules
# (Allow/Disallow) say where a crawler may go. The usage lines say what
# anyone who fetched a page may do with it. Search stays open: the search
# crawler keeps indexing, and the ai-train signal is the one it honours.

User-agent: *
Allow: /
Disallow: /account/
Disallow: /api/
Content-Signal: search=yes, ai-input=no, ai-train=no
Content-Usage: train-ai=n, ai-use=n

# A scraping intermediary that declares itself gets no access at all. It will
# ignore this, and the log line that proves it ignored this is the point.
User-agent: ScrapeFeed-Bot
Disallow: /
